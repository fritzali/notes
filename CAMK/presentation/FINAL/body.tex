	\section{Numerical Setup}

	\begin{frame}
		\frametitle{Black Hole Adaptation}
		\begin{itemize}
			\item<2-> Adapted Viscous Star Disk Setup from Numerical Atlas to Nonrotating SMBH
			\begin{itemize}
				\item<3-> Solved 2.5D on Spherical Grid in $r$ and $\theta$ with Axial \& Equatorial Symmetry
				\item<4-> Replaced Stellar Surface with Absorbing Diode Inner Boundary
				\item<5-> Replaced Newtonian with Paczyński–Wiita Potential to Reproduce ISCO
						  $$\Phi = -\frac{GM}{r - r_s}$$
				\item<6-> Measured Lengths and Times in Gravitational Units with $c$ for Velocities
						  \begin{align*}&&&& r_g = GM / c^2 && t_g = r_g / c &&&&\end{align*}
				\item<7-> Embedded Kluźniak–Kita Disk from $R = 6$ with $\epsilon = 0.1$ in Hydrostatic Corona
				\item<8-> Reinjected Analytic Disk at Outer Boundary and Blocked Coronal Inflow
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Radiation Physics}
		\begin{itemize}
			\item<2-> Enabled Nonrelativistic Two Moment Radiation Module with Full Speed of Light
			\begin{itemize}
				\item<3-> Fixed Physical Units by Black Hole Mass of \qty{1e8}{\solarmass} and Density Unit
						  \begin{align*}&&&& r_g \approx \qty{1.48e13}{\centi\meter} && t_g \approx \qty{493}{\second} &&
						  \rho_0 = \qty{5.8e-10}{\gram\per\centi\meter\cubed} &&&&\end{align*}
				\item<4-> Balanced Initial Viscous Heating and Radiative Cooling at $\alpha = 0.5$
				\item<5-> Included Kramers Absorption as well as Thomson Scattering Opacities
			\end{itemize}
			\item<6-> Added Effective Compton Opacity since PLUTO Treats Scattering as Coherent
					  $$\kappa_C = \kappa_\mathrm{es} \frac{4 k_B T_r^4}{m_e c^2 \left(T_g + T_r\right) \left(T_g^2 + T_r^2\right)}$$
			\vspace{-1.5ex}
			\begin{itemize}
				\item<7-> Reproduces Compton Exchange Rate and Prevents Gas Heating up to \qty{1e11}{\kelvin}
				\item<8-> Rate Capped above \qty{1e9}{\kelvin}, Decoupled above $0.9 \, c$ and Gas Floor at \qty{1e4}{\kelvin}
			\end{itemize}
		\end{itemize}
	\end{frame}

	\section{Disk Classifier}

	\begin{frame}
		\frametitle{Passive Tracer}
		\begin{itemize}
			\item<2-> Advected Tracer Switches Viscosity, Resistivity, Cooling and Opacities to Disk
					  $$\partial_t q + \mathbf{v} \cdot \nabla q = 0$$
			\vspace{-1.5ex}
			\begin{itemize}
				\item<3-> Stays at Initial Wedge Once Radiation Enabled while Disk Evolves
				\item<4-> Cause Not Isolated in Code, Only Restricted Physics Affected
			\end{itemize}
			\item<5-> Replaced Tracer by Classifier Built from Current State of Each Cell Alone
			\begin{itemize}
				\item<6-> Same Disk Condition as Initial Setup, Identical to Tracer at Start
				\item<7-> Robust to Restarts by Writing Classifier State with Every Output
			\end{itemize}
		\end{itemize}
	\end{frame}

				\item<5-> Multiplied Density and Keplerian Rotation Criteria with Logistic Function

				\item<6-> Updated References as $f$ and $1 - f$ Weighted Means via Moving Average with\break
						  Weight $w = 1\kern-0.5pt - \exp\left(-\Delta t / \tau\right)$ and Time Constant $\tau = 6 \, t_g$
			\end{itemize}

	\section{Discussion}

	\begin{frame}
		\frametitle{Limitations}
		\begin{itemize}
			\item<2-> Viscosity Uses Sound Speed of Initial Disk, Fixed in Time
			\begin{itemize}
				\item<3-> Removes Feedback Driving Thermal and Viscous Instability, so Stability No Test
			\end{itemize}
			\item<4-> Boundaries Set Gas Supply and May Cause Inflow Stream and Wave Period
			\begin{itemize}
				\item<5-> Mass Budget Open by \qty{3}{\percent} between $r = 2.5$ and \num{25}, \qty{18}{\percent} for Full Domain
			\end{itemize}
			\item<6-> Gas Temperatures Likely Too High for Setup to Be Fully Trusted
			\begin{itemize}
				\item<7-> Corona up to \qty{1e11}{\kelvin} with $k_B T$ about \num{46} Times Electron Rest Energy
				\item<8-> Neither Relativistic Compton Corrections nor Pair Production Included
			\end{itemize}
			\item<9-> Plunging Gas inside about Four $r_g$ Moves Superluminally in Pseudo Potential
			\item<10-> Axisymmetric Hydrodynamics Replaces MRI Turbulence by $\alpha$ Prescription
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Model Equations}
		\begin{itemize}
			\item Initial Disk Pressure and Density from Analytic Solution of Kluźniak \& Kita
				  \begin{align*}&&&& C = \frac{2}{5 \epsilon^2} \left(\frac{1}{r} - \frac{1 - 5 \epsilon^2 / 2}{R}\right) &&
				  p = \epsilon^2 C^{5/2} && \rho = C^{3/2} &&&&\end{align*}
			\item Corona in Hydrostatic Equilibrium with $\rho_c = 0.01$ in Code Units
				  \begin{align*}&&&& \rho = \rho_c r^{-3/2} && p = 0.4 \rho_c r^{-5/2} &&&&\end{align*}
			\item Shear Viscosity with Sound Speed of Initial Disk, Restricted by Classifier
				  \begin{align*}&&&& \eta_\nu = \frac{2}{3} \alpha \rho \epsilon^2 C R^{3/2} f && \nu = \frac{2}{3} \alpha \epsilon^2 R^{1{\kern-0.5pt}/2} &&&&\end{align*}
			\item Density Floor at \num{5e-7} in Code Units with Constant Sound Speed and Momentum
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Opacity Model}
		\begin{itemize}
			\item Kramers Free Free and Bound Free Absorption with Guillotine Factor
				  \begin{align*}&&&& \kappa_K = \left(K_\mathrm{ff} + K_\mathrm{bf}\right) \rho \, T_g^{-7/2} && \kappa_\mathrm{es} = 0.2 \left(1 + X\right) &&&&\end{align*}
			\item Compton Exchange Held at Cap above $T_\mathrm{max} = \qty{1e9}{\kelvin}$
				  $$\kappa_C = \kappa_\mathrm{es} \frac{4 k_B T_r^4 \left(T_\mathrm{max} - T_r\right)}{m_e c^2 \left(T_g^4 - T_r^4\right)}$$
			\item Gated by Classifier and Decoupled near Speed of Light, Thomson Scattering on All Gas
				  \begin{align*}&&&& f_d = \max\left(\frac{f - 0.05}{0.95}, \, 0\right) &&
				  g_v = \left[1 + \exp\left(\frac{\beta - 0.9}{0.03}\right)\right]^{-1} &&&& \\
				  &&&& \kappa_\mathrm{abs} = g_v \,f_d \left(\kappa_K + \kappa_C\right) &&
				  \kappa_\mathrm{sc} = g_v \left(\kappa_\mathrm{es} - f_d \kappa_C\right) &&&&\end{align*}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Classifier Details}
		\begin{itemize}
			\item Corona Reference as $1 - f$ and Disk Reference as $f$ Weighted Mean Density per Bin
			\begin{itemize}
				\item Disk Estimate Counts Only above One Thousandth of Bin Volume
				\item Bins without Valid Estimate Copy Nearest Valid Bin
			\end{itemize}
			\item Guards Added after Inversion in First Version
			\begin{itemize}
				\item Span below \qty{0.5}{\dex} Copies Both References from Nearest Sufficient Bin
				\item Disk Reference Raised to Minimum Span if No Such Bin Exists
				\item Fixed Weight \num{0.05} per Step Replaced by Time Constant $\tau$
			\end{itemize}
			\item Optional Geometric Mode Independent of Classification
			\begin{itemize}
				\item Disk Reference from Maximum Density, Corona from Geometric Mean above $z / R = 0.5$
			\end{itemize}
		\end{itemize}
	\end{frame}
