\tikzset{declare function={
  dx(\X)=\cx+\X;
  dy(\Y)=\cy+\q*\Y;
  % logarithmic spiral, radius at angle t in radians, and the width of an arm tapering to both ends
  arm(\t)=0.55*exp(0.3*\t);
  armw(\t)=0.4*sin(180*\t/6.3);
  % magnetic field line along the same spiral, smooth except for a short rippled stretch around
  % t = c, where the instability grows
  bfr(\t,\c)=arm(\t)+0.15*exp(-((\t-\c)/0.26)^2)*sin(1150*(\t-\c));
}}

% supernova shell of radius #3 at (#1,#2), a transparent bubble sitting in the disk with its centre a
% height #4 above the disk plane, the part below the plane seen dimmed through the gas of the disk,
% and the circle where it cuts the plane, its ellipse touching the outline of the sphere exactly, at
% the angles t on the ellipse and p on the outline worked out from the projection
\newcommand{\snshell}[4]{\begin{scope}[shift={(#1,#2)}]
  \pgfmathsetmacro{\hc}{#4*sqrt(1-\q*\q)}\pgfmathsetmacro{\ri}{sqrt(#3*#3-#4*#4)}
  \pgfmathsetmacro{\tt}{asin(-\q*#4/(\ri*sqrt(1-\q*\q)))}
  \pgfmathsetmacro{\pp}{atan2(\q*\ri*sin(\tt)-\hc,\ri*cos(\tt))}
  \begin{scope}
    \clip ({\ri*cos(180-\tt)},{\q*\ri*sin(180-\tt)})
      arc[start angle=180-\tt,end angle=360+\tt,x radius=\ri,y radius=\q*\ri]
      arc[start angle=\pp,end angle=180-\pp,radius=#3] -- cycle;
    \fill[shellc,path fading=shellfade] (0,\hc) circle (#3);
  \end{scope}
  \begin{scope}
    \clip ({\ri*cos(180-\tt)},{\q*\ri*sin(180-\tt)})
      arc[start angle=180-\tt,end angle=360+\tt,x radius=\ri,y radius=\q*\ri]
      arc[start angle=\pp,end angle=-180-\pp,radius=#3] -- cycle;
    \fill[shellc,path fading=shellfade,opacity=0.4] (0,\hc) circle (#3);
  \end{scope}
  \draw[shellc!70,line width=0.35pt] (0,0) ellipse[x radius=\ri,y radius=\q*\ri];
  \end{scope}}

% one snapshot #6 of the box, turned away from the viewer: drawn in its own frame from (0,0) to
% (sL,sL), mapped onto a quadrilateral in a small picture of its own, which is then placed with the
% lower left corner of the frame at (#1,#2), its right side taller than its left by fd at top and
% bottom alike, as if the right side were nearer and the snapshot at the height of the viewer, with
% its gas turned by #3 degrees, moved by #5, and an extra patch of gas at #4, and its stirring arrows
% #6 from gensnap.py, so the forcing changes gradually from one snapshot to the next
% the mapping takes (x,y) in the frame of a snapshot to the page: x across to the width sw, y up to
% the height sH of the far left side, and both sides opening by fd towards the near right side,
% worked in whole scaled points with e-TeX, as pgfmath would run out of range on these products
\makeatletter
\def\framemap{%
  \pgf@xa=\pgf@x\pgf@ya=\pgf@y
  \pgf@yb=\dimexpr\pgf@xa*\number\pgf@ya/\fLsp\relax
  \pgf@x=\dimexpr\fxsp sp+\pgf@xa*\fwsp/\fLsp\relax
  \pgf@y=\dimexpr\fysp sp+\pgf@xa*\fkdsp/\fLsp+\pgf@ya*\fHsp/\fLsp+\pgf@yb*\ftdsp/\fLsp\relax}
% the lengths the mapping needs, in scaled points, for a snapshot with its frame corner at (#1,#2)
\def\setframe#1#2{%
  \pgfmathsetlength\pgf@xa{(#1)*1cm}\edef\fxsp{\number\pgf@xa}%
  \pgfmathsetlength\pgf@xa{(#2)*1cm}\edef\fysp{\number\pgf@xa}%
  \pgfmathsetlength\pgf@xa{\sL*1cm}\edef\fLsp{\number\pgf@xa}%
  \pgfmathsetlength\pgf@xa{\sw*1cm}\edef\fwsp{\number\pgf@xa}%
  \pgfmathsetlength\pgf@xa{\sH*1cm}\edef\fHsp{\number\pgf@xa}%
  \pgfmathsetlength\pgf@xa{-\fd*1cm}\edef\fkdsp{\number\pgf@xa}%
  \pgfmathsetlength\pgf@xa{2*\fd*1cm}\edef\ftdsp{\number\pgf@xa}}
\makeatother
% each snapshot is mapped near the origin of a small picture of its own, as the nonlinear mapping
