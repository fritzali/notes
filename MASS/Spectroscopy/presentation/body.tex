	\section{Appendix}

	\begin{frame}
		\frametitle{Appendix}
		\begin{itemize}
			\item Total Contained Thermal Energy $$E_t \propto nT$$
			\item Radiative Cooling Rate $$\dot{E}_c \propto n^2 \Lambda_c(T) \propto n^2 T^{-1\kern-0.5pt/2}$$
			\item Characteristic Cooling Time $$t_c \equiv \frac{E_t}{\dot{E}_c} \propto T^{1\kern-0.5pt/2} n^{-1}$$
			\item Bremsstrahlung Emissivity $$\epsilon \propto n_e n_i Z^2 T^{-1\kern-0.5pt/2} g_{f\kern-0.5ptf}(T, E) \, e^{-E/kT}$$
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Appendix}
		\begin{itemize}
			\item Salpeter Initial Stellar Mass Function $$\xi(M_*) = \frac{dN_*}{dM_*} \propto M_*^{-\alpha}$$
			\item Hydrostatic Equilibrium Condition $$\nabla P = g\rho = -\frac{GM\rho}{r^2}$$
			\item Pressure Relation $$P = nkT$$
			\item Entropy Relation $$K = \frac{kT}{n^{2/3}}$$
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Appendix}
		\begin{itemize}
			\item Density Profile \begin{align*} &&&& n_{NCC} = n_0 \left( 1 + \frac{r^2}{r_c^2} \right)^{-3\beta / 2}
				  && n_{CC} = \sqrt{n_1^2 + n_2^2} &&&& \end{align*}
			\item Temperature Profile \begin{align*} && T = T_0 \frac{x_\text{cool} + T_\text{min} / T_0}{x_\text{cool} + 1}
				  \frac{(r / r_t)^{-a}}{\bigl( 1 + (r / r_t)^{b} \bigr)^{c/b}} && x_\text{cool} = (r / r_\text{cool})^{a_\text{cool}} && \end{align*}
			\item Pressure Profile $$P = P_0 \left( (C_\ell \, x_\ell)^c (1 + \bigl(C_\ell \, x_\ell)^a \bigr) \right)^{-(b-c)/a}$$
			\item Entropy Profile \begin{align*} &&&&&& K = K_0 + K_\ell \, x_\ell^\gamma && x_\ell = r / R_\ell &&&&&& \end{align*}
		\end{itemize}
	\end{frame}
