	\subsection{Definitions}

	\begin{frame}
		\frametitle{Frequentist View}
		\begin{itemize}
			\item<2-> \itshape{With Prior Knowledge:} out of all $n$ different but equally likely possible outcomes of an event,
													  there are $k$ that have a property $A$ which is therefore measured with probability
													  $$P(A) = \frac{k}{n}$$
			\item<3-> \itshape{Without Prior Knowledge:} out of $n$ independent measurements the outcome $A$ occurs $k$ times, defining
														 $$P(A) = \underset{n \to \infty}{\symup{lim}} \frac{k}{n}$$
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Bayesian View}
		\begin{itemize}
			\item<2-> $P(A|B)$ quantifies the plausibility of an assumption $A$ under known information given by $B$
			\item<3-> \itshape{Kolmogorov Axioms} underpin the rules for calculating with probabilities
			\item<4-> $P(A,B) = P(A)\,P(B|A)$ quantifies the probability of measuring $A$ and $B$ simultaneously, and from assuming
					  $P(A,B) = P(B,A)$ we find
					  \visible<5->{\begin{center}\begin{minipage}{0.3\textwidth}\begin{block}{\centering Bayes' Theorem}
						  \vspace{-1ex}$$P(A|B) = \frac{P(B|A)\,P(A)}{P(B)}$$\vspace{-1.5ex}
					  \end{block}\end{minipage}\end{center}}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Continuous Distributions}
		\begin{itemize}
			\item<2-> \itshape{Cumulative Distribution Function:} $\qquad F(x) = P(X \leq x)$
			\item<3-> \itshape{Probability Density Function:} $\qquad f(x) = \dfrac{d}{dx} F(x)$
					  \visible<4->{$$P(a \leq X \leq b) = \!\!\int_a^b f(x) \, dx = F(b) - F(a)$$}
			\item<5-> \itshape{Normalization:} \visible<6->{$\qquad\displaystyle{1 = \!\!\int_{-\infty}^{+\infty} f(x)\,dx}$}
											   \visible<7->{$\qquad\displaystyle{\underset{x \to -\infty}{\symup{lim}} F(x) = 0}$}
											   \visible<7->{$\qquad\displaystyle{\underset{x \to +\infty}{\symup{lim}} F(x) = 1}$}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Expectation Values}
			\visible<2->{$$E[h(x)] = \!\!\int_{-\infty}^{+\infty} h(x) \,f(x) \,dx$$\vspace{4ex}}
		\begin{itemize}
			\item<3-> \itshape{Mean} or first raw moment: $\displaystyle{\mu = E[X] = \!\!\int_{-\infty}^{+\infty} xf(x) \,dx}$
			\item<4-> \itshape{Variance} or second raw moment: $\displaystyle{\sigma^2 = E\bigl[(X - E[X])^2\bigr] =
															   \!\!\int_{-\infty}^{+\infty} \bigl(x - E[x]\bigr)^2 f(x) \,dx}$
		\end{itemize}
	\end{frame}

	\subsection{Notation}

	\begin{frame}
		\frametitle{Drawing Numbers}
			\visible<2->{$$x \sim \mathcal{U}(a, b) \qquad\qquad y \sim \mathcal{N}(\mu, \sigma)$$\vspace{3ex}}
		\begin{itemize}
			\item<3-> \itshape{Uniform Distribution:} $\displaystyle{\mathcal{U}(a,b) = f(x|a,b) =
													   \begin{cases}
														   \frac{1}{b-a} & a\leq x\leq b \\
														   0 & \symup{otherwise}
													   \end{cases}}$
			\item<4-> \itshape{Gaussian Distribution:} $\displaystyle{\mathcal{N}(\mu,\sigma) = f(y|\mu,\sigma) =
														\frac{1}{\sqrt{2\pi \sigma^2}} \symup{exp}\left(-\frac{1}{2}
														\left(\frac{y - \mu}{\sigma}\right)^2\right)}$
		\end{itemize}
	\end{frame}

		\frametitle{Motivation}
		\begin{itemize}
			\item<2-> \itshape{Model Optimisation:} finding the best possible model parameters
			\begin{itemize}
				\item<3-> maximising objective function \itshape{(Fitting)}
				\item<4-> minimising loss function \itshape{(Machine Learning)}
			\end{itemize}
			\item<5-> \itshape{Example:} modified blackbody spectrum for galactic dust emission (M. Juvela, {\itshape{A\&A}} 2023)
									\only<6|handout:0>{
									$$I_\nu(T, \beta) = B_\nu(T) \left( 1 - e^{-\tau_\nu (\beta)}\right) \approx B_\nu(T) \tau_\nu (\beta)
									= B_\nu(T) \Sigma \kappa_\nu(\beta) = \frac{2h\Sigma\kappa_0}{c^2{\nu_0}^\beta} \frac{\nu^{3+\beta}}
									{\symup{exp}\left(\frac{h\nu}{kT}\right) - 1}\vspace{-3ex}$$
									}\only<7->{
									$$I_\nu(T,\beta) \equiv \vphantom{\frac{2h\Sigma\kappa_0}{c^2{\nu_0}^\beta}}
									\frac{\nu^{3+\beta}}{\symup{exp}\left(\frac{h\nu}{kT}\right) - 1}$$\vspace{-3ex}
									}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Example}
		\visible<2->{$$I_\nu(T,\beta) = \vphantom{\frac{2h\Sigma\kappa_0}{c^2{\nu_0}^\beta}}
					 \frac{\nu^{3+\beta}}{\symup{exp}\left(\frac{h\nu}{kT}\right) - 1}$$\vspace{2ex}}
		\begin{itemize}
			\item<3-> \itshape{Objective:} optimise $\theta \equiv (T, \beta)$ such that the weighted squared error is minimal
			\item<4-> \itshape{Assumptions:}
			\begin{itemize}
				\item<5-> independent measurements generated by $I_\nu(\theta)$ with normally distributed uncertainty
				\item<6-> infinitesimally small error in $\nu$ so that we only need $\sigma_\nu$ in $I_\nu$ direction
				\item<7-> suppose we have no prior information on the parameter distribution
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}[noframenumbering]
		\frametitle{Example}
		\begin{itemize}
			\item<2-> \itshape{Likelihood:} \only<2|handout:0>{
											$$L(I_\nu|\nu, \sigma_\nu\, , \theta) = \prod_{i=1}^N\frac{1}{\sqrt{2\pi \sigma_{\nu,i}^2}}
											\symup{exp}\left(-\frac{\left(I_{\nu,i} - I_\nu(\theta)\right)^2}{2\sigma_{\nu,i}^2}\right)$$}
											\only<3|handout:0>{$$\vphantom{\prod_{i=1}^N\frac{1}{\sqrt{2\pi \sigma_{\nu,i}^2}}
											\symup{exp}\left(-\frac{\left(I_{\nu,i} - I_\nu(\theta)\right)^2}{2\sigma_{\nu,i}^2}\right)}
											\symup{log\,} L (I_\nu|\nu, \sigma_\nu\, , \theta) = -\frac{N}{2}\symup{log}(2\pi) -
											\sum_{i=1}^N \sigma_{\nu,i} - \frac{1}{2}\sum_{i=1}^N \frac{\left(I_{\nu,i} - I_\nu(\theta)\right)^2}
											{\sigma_{\nu,i}^2}$$}
											\only<4->{$$\vphantom{\prod_{i=1}^N\frac{1}{\sqrt{2\pi \sigma_{\nu,i}^2}}
											\symup{exp}\left(-\frac{\left(I_{\nu,i} - I_\nu(\theta)\right)^2}{2\sigma_{\nu,i}^2}\right)}
											-\symup{log\,} L (I_\nu|\nu, \sigma_\nu\, , \theta) \propto \chi^2(\theta) = \sum_{i=1}^N \frac{\left(I_{\nu,i}
											- I_\nu(\theta)\right)^2}{\sigma_{\nu,i}^2}$$}
			\item<5-> \itshape{Posterior:} \only<5|handout:0>{
											$$P(\theta |I_\nu\, , \nu, \sigma_\nu) \propto L(I_\nu|\nu, \sigma_\nu, \theta)$$\vspace{-1ex}}
											\only<6|handout:0>{
											$$\symup{log\,}P(\theta |I_\nu\, , \nu, \sigma_\nu) \propto \symup{log\,}L(I_\nu|\nu, \sigma_\nu, \theta)$$
											\vspace{-1ex}}
											\only<7->{
											$$\symup{log\,}P(\theta |I_\nu\, , \nu, \sigma_\nu) \propto -\chi^2(\theta)$$\vspace{-1ex}}
			\item<only@8-> \itshape{Objective:} minimise the $\chi^2(\theta)$ measure to maximise the posterior probability
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Parameter Grid}
		\begin{itemize}
			\item<2-> \itshape{Idea:} in our $D=2$ dimensional parameter space, we can create an $N \times N$ grid of $(T, \beta)$ points
									  and evaluate the scalar function at each one
			\item<3-> \itshape{Preparation:} for this as well as some of the following methods, it is a good idea to normalize the variation
											 of each individual parameter
				\begin{itemize}
					\item<4-> from the literature we expect $(T,\beta) \in [10, 25] \times [1.5, 2.0]$ as constraints
					\item<5-> resize these intervals by defining $T \mapsto \tilde{T} = T / \, 30$ with $I_\nu(T, \beta) \rightarrow I_\nu(\tilde{T}, \beta)$
				\end{itemize}
			\item<6-> \itshape{Problem:} as $D$ increases we have $N^D$ gridpoints, and with more complex parameter distributions, this procedure
										 quickly becomes intractable due to computational limitations
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{General Case}
		\begin{itemize}
			\item<2-> \itshape{Optimisation Algorithms:} intelligently move through parameter space towards optimum
			\item<3-> \itshape{Common Attributes:}
				\begin{enumerate}
					\item<4-> choose initialisation point
					\item<5-> evaluate objective function or its derivatives to estimate local curvature
					\item<6-> propose new step in parameter space
					\item<7-> decide whether proposal is a good step and update position accordingly
					\item<8-> repeat until some predefined cutoff is satisfied
				\end{enumerate}
			\item<9-> \itshape{Interactive Visualisations:} check out Ben Frederickson$^{\href{https://www.benfrederickson.com/numerical-optimization/}{1}}$
					  and Andy Casey$^{\href{https://andycasey.github.io/teaching/phs5000/4/}{\,2}}$ to see some nice examples
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Nelder-Mead}
		\begin{itemize}
			\item<2-> \itshape{Requirements:} only objective function, no information on gradients
			\item<3-> \itshape{Algorithm:}
				\begin{enumerate}
					\item<4-> initialize simplex of $d+1$ points in $d$ dimensional parameter space
					\item<5-> rank points in order of values $f(x_1) \leq \cdots \leq f(x_d) \leq f(x_{d+1})$
					\item<6-> compute centroid $x_0$ of $x_1\, , \ldots , x_d$
					\item<7-> perform a bunch of conditional evaluations between $x_0$ and $x_{d+1}$
					\item<8-> reflect, expand, contract or shrink based on results to move points downhill and repeat
				\end{enumerate}
			\item<9-> \itshape{Notes:}
				\begin{itemize}
					\item<10-> extrapolates function to slowly move geometric blob through parameter space
					\item<11-> zeroth order method since no derivatives are used
				\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}[noframenumbering]
		\frametitle{Nelder-Mead}
		\begin{itemize}
			\item<2-> \itshape{Conditions:}
				\begin{enumerate}
					\item<3-> \itshape{Reflection:}
					\begin{itemize}
						\item<4-> compute reflected point $x_r = x_0 + \alpha(x_0 - x_{d+1})$ with $\alpha > 0$
						\item<5-> if $x_r$ better than second worst $x_d$ but not better than best $x_1$ with $f(x_1) \leq f(x_r) < f(x_d)$ then
								  obtain new simplex by replacing $x_{d+1}$ with $x_r$ and repeat from start
						\item<6-> continue to the next step instead if $x_r$ is the best point with $f(x_r) < f(x_1)$
						\item<7-> skip the next skip and go to the one after it if $x_r$ is not better than the second worst $x_d$ with
								  $f(x_r) \geq f(x_d)$
					\end{itemize}
				\end{enumerate}
		\end{itemize}
	\end{frame}

	\begin{frame}[noframenumbering]
		\frametitle{Nelder-Mead}
		\begin{itemize}
			\item<2-> \itshape{Conditions:}
				\begin{enumerate}
					\setcounter{enumi}{1}
					\item<3-> \itshape{Expansion:}
					\begin{itemize}
						\item<4-> compute expanded point $x_e = x_0 + \gamma(x_r - x_0)$ with $\gamma > 1$
						\item<5-> if $x_e$ is better than $x_r$ with $f(x_e) < f(x_r)$ then obtain new simplex by replacing the worst $x_{d+1}$
								   with $x_e$ and repeat from start
						\item<6-> else obtain new simplex by replacing the worst $x_{d+1}$ with $x_r$ and repeat from start
					\end{itemize}
					\item<7-> \itshape{Contraction:}
					\begin{itemize}
						\item<8-> if $x_r$ is better than the worst $x_{d+1}$ with $f(x_r) < f(x_{d+1})$ then compute the outside contracted point
								  $x_c = x_0 + \rho (x_r - x_0)$ where $0 < \rho \leq 0.5$
						\item<9-> for $x_c$ better than $x_r$ with $f(x_c) < f(x_r)$ obtain new simplex by replacing worst $x_{d+1}$ with $x_c$
								  and repeat from start, otherwise go to the next step
					\end{itemize}
				\end{enumerate}
		\end{itemize}
	\end{frame}

	\begin{frame}[noframenumbering]
		\frametitle{Nelder-Mead}
		\begin{itemize}
			\item<2-> \itshape{Conditions:}
			\begin{enumerate}
				\setcounter{enumi}{2}
				\item<3-> \itshape{Contraction:}
				\begin{itemize}
					\item<4-> if $x_r$ is not better than the worst $x_{d+1}$ with $f(x_r) \geq f(x_{d+1})$ then compute the inside contracted point
							  $x_c = x_0 + \rho (x_{d+1} - x_0)$ where $0 < \rho \leq 0.5$
					\item<5-> for $x_c$ better than the worst $x_{d+1}$ with $f(x_c) < f(x_{d+1})$ obtain new simplex by replacing worst $x_{d+1}$
							  with $x_c$ and repeat from start, otherwise go to next step
				\end{itemize}
				\item<6-> \itshape{Shrink:}
				\begin{itemize}
					\item<7-> obtain new simplex by replacing all points except the best $x_1$ with $x_i = x_1 + \sigma(x_i - x_1)$ and
							  coefficient $0 < \sigma < 1$ before repeating from start
				\end{itemize}
			\end{enumerate}
			\item<8-> \itshape{Standard Values:} $\alpha = 1 \;\&\; \gamma = 2 \;\&\; \rho = \sigma = 0.5$
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Random Descent}
		\begin{itemize}
			\item<2-> \itshape{Requirements:} only objective function, no information on gradients
			\item<3-> \itshape{Algorithm:}
				\begin{enumerate}
					\item<4-> start from arbitrary inital value $x_0$
					\item<5-> draw a proposed shift from some chosen probability distribution $X \sim g(x)$
					\item<6-> propose updated position $\hat{x}_{k+1} = x_k + X$
					\item<7-> accept proposal $x_{k+1} = \hat{x}_{k+1}$ only if $f(\hat{x}_{k+1}) < f(x_{k+1})$ and repeat
				\end{enumerate}
			\item<8-> \itshape{Notes:}
				\begin{itemize}
					\item<9-> traverses parameter space via modified random walk, can be very slow
					\item<10-> zeroth order method since no derivatives are used
					\item<11-> improve by considering last step and optimizing step range
				\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Gradient Descent}
		\begin{itemize}
			\item<2-> \itshape{Requirements:} first derivatives of the objective function, can be estimated numerically
			\item<3-> \itshape{Algorithm:}
				\begin{enumerate}
					\item<4-> initialize at some $x_0$
					\item<5-> update current point with $x_{k+1} = x_k - \alpha\nabla f(x_k)$
					\item<6-> iterate until convergence $|x_{k+1} / x_k - 1| < \epsilon$ or maximum step count is reached
				\end{enumerate}
			\item<7-> \itshape{Notes:}
				\begin{itemize}
					\item<8-> moves along direction of steepest descent at each point with fixed learning rate $\alpha$
					\item<9-> first order method since only the gradient is used
					\item<10-> improve by performing line search for $\alpha$ at every step
				\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Backtracking Line Search}
		\begin{itemize}
			\item<2-> \itshape{Requirements:} first derivatives of the objective function, can be estimated numerically
			\item<3-> \itshape{Algorithm:}
				\begin{enumerate}
					\item<4-> start from some fixed large $\alpha_0$ for each step $k$ of the gradient descent
					\item<5-> set $\alpha_{j+1} = \tau\alpha_j$ until $f\left(x_k\right) - f\left(x_k - \alpha_j\nabla f(x_k)\right)
							  \geq \kappa\alpha_j\left(\nabla f(x_k)\right)^2$ where $\tau, \kappa \in (0,1)$
					\item<6-> use last value to set learning rate $\alpha = \alpha_j$ for next step
				\end{enumerate}
			\item<7-> \itshape{Notes:}
				\begin{itemize}
					\item<8-> ensures $f(x_{k+1}) < f(x_k)$ such that the function steadily descreases with increasing $k$
					\item<9-> favours secant between $x_k$ and $x_{k+1}$ steeper or almost as steep as tangent $\nabla f(x_k)$
					\item<10-> robust convergence guarantees, larger step size for flat areas avoids saddle points
					\item<11-> first order method since only the gradient is used
					\item<12-> typical settings are $\tau = \kappa = 0.5$ (L. Armijo, {\itshape{Pac. J. Math.}} 1966)
				\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Conjugate Gradient Descent}
		\begin{itemize}
			\item<2-> \itshape{Requirements:} first derivatives of the objective function, can be estimated numerically
			\item<3-> \itshape{Algorithm:}\vspace{-1.25ex}
				\begin{enumerate}
					\item<4-> calculate search direction $s_k = \nabla f(x_k) + \beta s_{k-1}$ where $\beta = \dfrac{\nabla f(x_k) \left(\nabla f(x_k) -
							  \nabla f(x_{k-1})\right)}{\left(\nabla f(x_{k-1})\right)^2}$\vspace{-1.25ex}
					\item<5-> update point with $x_{k+1} = x_k + \alpha s_k$ after a line search for $\alpha$\vspace{0.25ex}
					\item<6-> iterate until convergence $|x_{k+1} / x_k - 1| < \epsilon$ or maximum step count is reached
				\end{enumerate}
			\item<7-> \itshape{Notes:}
				\begin{itemize}
					\item<8-> uses gradients from previous evaluations to estimate curvature
					\item<9-> smooths oscillatory behaviour of naive gradient descent, especially in tight valleys with nearby
							  gradients almost orthogonal to search direction
					\item<10-> first order method since only the gradient is used
				\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Adaptive Moment Estimation}
		\begin{itemize}
			\item<2-> \itshape{Requirements:} first derivatives of the objective function, can be estimated numerically
			\item<3-> \itshape{Algorithm:}
				\begin{enumerate}
					\item<4-> initialize position $x_0$ as well as moving averages $m_0 = 0$ and $v_0 = 0$\vspace{0.5ex}
					\item<5-> set $m_{k+1} = \beta_1 m_k + (1 - \beta_1) \nabla f(x_k)$ and
							  $v_{k+1} = \beta_2 v_k + (1 - \beta_2) \left(\nabla f(x_k)\right)^2$\vspace{0.5ex}
					\item<6-> correct estimator bias with $\tilde{m}_k = m_{k+1}(1 - \beta_1^k)^{-1}$ and $\tilde{v}_k = v_{k+1}(1 - \beta_2^k)^{-1}$
					\item<7-> update the position $x_{k+1} = x_k - \alpha \tilde{m}_k \left(\sqrt{\tilde{v}_k} + \varepsilon\right)^{-1}$ and repeat
				\end{enumerate}
			\item<8-> \itshape{Notes:}
				\begin{itemize}
					\item<9-> point behaves as iterative analogue to ball rolling in landscape with friction
					\item<10-> hyperparameters $\beta_1, \beta_2 \in (0,1)$ are forgetting factors for previous gradients, and $\varepsilon \ll 1$ 
					\item<11-> first order method since only the gradient is used
				\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Newton Method}
		\begin{itemize}
			\item<2-> \itshape{Requirements:} first and second derivatives of the objective function, can be estimated numerically
			\item<3-> \itshape{Algorithm:}
				\begin{enumerate}
					\item<4-> choose some initial $x_0$\vspace{1ex}
					\item<5-> calculate $\nabla f(x_k) = \left(\frac{\partial f(x_k)}{\partial z_i}\right)_{i = 1,\ldots,d}$ and
							  $H_{f(x_k)} = \left(\frac{\partial^2 f(x_k)}{\partial z_i \partial z_j}\right)_{i,\,j = 1,\ldots,d}$
					\item<6-> update the parameters $x_{k+1} = x_k - \alpha H_{f(x_k)}^{-1} \nabla f(x_k)$ with dampening $\alpha$
				\end{enumerate}
			\item<7-> \itshape{Notes:}
				\begin{itemize}
					\item<8-> generalises Newton-Raphson method for finding roots of derivative to multiple dimensions
					\item<9-> finds critical point, no matter if maximum, minimum or saddle point, sensitive to initialisation
					\item<10-> second order method since both gradient and Hessian are used
					\item<11-> related Gauss-Newton algorithm employs Jacobian instead of Hessian
				\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Advanced Algorithms}
		\begin{itemize}
			\item<2-> \itshape{Traits:} quasi second order methods, interpolate between optimsers, various curvature estimations

		\begin{itemize}
			\item<2-> \itshape{Optimisation:} finds a point estimate of optimal model parameters
			\item<3-> \itshape{Monte Carlo Sampling:} dynamically explores the parameter space distribution
			\begin{itemize}
				\item<4-> quote asymmetric percentiles
				\item<5-> estimate expectation values by integrating
				\item<6-> uncertainty propagation to the model function
				\item<7-> simulate complex measurement processes and quantify the total convolution
			\end{itemize}
		\end{itemize}

	\begin{frame}
		\frametitle{Markov Chain}
		\begin{itemize}
			\item<2-> \itshape{Properties:}
			\begin{itemize}
				\item<3-> sequence generated by stochastic transitions
				\item<4-> transition probabilities depend only on the initial and final state
			\end{itemize}
			\item<5-> \itshape{Detailed Balance:}
			\begin{itemize}
				\item<6-> any chain belonging to a Markov Process represents a reversible sequence of events
				\item<7-> when walking along a stationary distribution $f(x)$ the following must hold for every
						  transition probability $M_{i \to j}$ from state $x_i$ to $x_j$
						  $$M_{i \to j} f(x_i) = M_{j \to i} f(x_j)$$
				\item<8-> then the average behaviour of this single process reproduces that over the entire ensemble
			\end{itemize}
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{Monte Carlo}
		\begin{itemize}
			\item<2-> \itshape{Randomness:}
			\begin{itemize}
				\item<3-> each proposed next step is accepted or discarded based on a probabilistic condition
				\item<4-> if the uncerlying distribution $f(x_{k+1})$ is larger at the new position $x_{k+1}$ it is updated
				\item<5-> else it is based on a random choice with acceptance being less likely for smaller $f(x_{k+1})$
			\end{itemize}
			\item<6-> \itshape{Rules:}
			\begin{itemize}
				\item<7-> the proposal distribution is allowed to violate detailed balance only if it is corrected in the
						  decision between acceptance and rejection
				\item<8-> unlike the underlying posterior that we sample, the proposal must be properly normalized
			\end{itemize}
		\end{itemize}
	\end{frame}

	\subsection{Algorithms}

	\begin{frame}
		\frametitle{Markov Chain Monte Carlo}
		\begin{itemize}
			\item<2-> \itshape{Metropolis:}
			\begin{itemize}
				\item<3-> a proposal $x_j$ is drawn from a fixed step size distribution such as a normal or uniform one
						  that is symmetric in each coordinate $g(x_j|x_i) = g(x_i|x_j)$\vspace{0.5ex}
				\item<4-> this satisifies detailed balance $M_{i \to j}f(x_i) = M_{j \to \i}f(x_i)$ with
						  $$M_{i \to j} = \symup{min}\left(1\, , \frac{f(x_j)}{f(x_i)}\right)$$
				\item<5-> generate a random number $\xi \sim \mathcal{U}(0,1)$ and decide according to
						  $$x_{i+1} = \begin{cases}x_j & \xi \leq M_{i \to j} \\ x_i & \xi > M_{i \to j}\end{cases}$$
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}[noframenumbering]
		\frametitle{Markov Chain Monte Carlo}
		\begin{itemize}
			\item<2-> \itshape{Metropolis-Hastings:}
			\begin{itemize}
				\item<3-> a proposal $x_j$ is drawn from an asymmetric distribution $g(x_j|x_i) \neq g(x_i|x_j)$\vspace{0.5ex}
				\item<4-> this itself violates detailed balance but is fixed by $M_{i \to j}f(x_i)g(x_j|x_i) = M_{j \to \i}f(x_j)g(x_i|x_j)$ with
						  $$M_{i \to j} = \symup{min}\left(1\, , \frac{f(x_j)g(x_i|x_j)}{f(x_i)g(x_j|x_i)}\right)$$
				\item<5-> generate a random number $\xi \sim \mathcal{U}(0,1)$ and decide according to
						  $$x_{i+1} = \begin{cases}x_j & \xi \leq M_{i \to j} \\ x_i & \xi > M_{i \to j}\end{cases}$$
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}[noframenumbering]
		\frametitle{Markov Chain Monte Carlo}
		\begin{itemize}
			\item<2-> \itshape{Comparison:}
			\begin{itemize}
				\item<3-> the Metropolis-Hastings is more flexible than the Metropolis algorithm since it can use asymmetric proposal distributions
				\item<4-> asymmetric proposals allow efficient handling of bounded parameter spaces
				\item<5-> proposal distribution shapes can be tailored to the sampled posterior to speed up exploration of the space and achieve
						  good convergence
				\item<6-> realistic cases, especially when priors are included, are generally not symmetric or unbounded, therefore benefitting from
						  the inbuilt bias correction to ensure correct sampling of the target distribution
			\end{itemize}
		\end{itemize}
	\end{frame}
	
	\begin{frame}
		\frametitle{Advanced Methods}
		\begin{itemize}
			\item<2-> \itshape{Affine-Invariant:}
			\begin{itemize}
				\item<3-> multiple walkers that spread out into parameter space controlled by stretch parameter
				\item<4-> biased proposals are made along a line connecting randomly paired walkers and corrected in the decision step
				\item<5-> very efficient at sampling correlated parameters due to reduced hyperparameters
				\item<6-> suitable in low dimensional parameter spaces where no gradient information is known
			\end{itemize}
		\end{itemize}
	\end{frame}
	
	\begin{frame}[noframenumbering]
		\frametitle{Advanced Methods}
		\begin{itemize}
			\item<2-> \itshape{Adjusted Langevin:}
			\begin{itemize}
				\item<3-> uses gradient of logarithmic likelihood or score function as a drift force pushing samples towards higher probability
				\item<4-> a small Gaussian noise term is added to act as a random force to allow exploration of the distribution
				\item<5-> bias is corrected with a Metropolis-Hastings decision step
				\item<6-> application in gradient estimation and diffusion lerning
			\end{itemize}
		\end{itemize}
	\end{frame}
	
	\begin{frame}[noframenumbering]
		\frametitle{Advanced Methods}
		\begin{itemize}
			\item<2-> \itshape{Hamilton:}
			\begin{itemize}
				\item<3-> treat target function as potential energy, introduce random momentum term for kinetic energy
				\item<4-> definition in canonical distribution ensures separability so that $x$ and $p$ are independent
				\item<5-> for each step, accurately integrate momentum some set amount of steps by using the gradient, conditioned
						  by the Hamiltonian or total energy remaining constant
				\item<6-> check proposal with Metropolis decision metric to account for numerical integration errors
				\item<7-> permits very long directed jumps that drastically reduce the correlation of the chain
				\item<8-> efficient for many dimensions with very high acceptance ratios
			\end{itemize}
		\end{itemize}
	\end{frame}

	\subsection{Tuning \& Diagnosing}

	\begin{frame}
		\frametitle{Sampling Efficiency}
		\begin{itemize}
			\item<2-> \itshape{Low Acceptance Rate:} $\lesssim30\%$
			\begin{itemize}
				\item<3-> only small fraction of significant part of target explored
				\item<4-> few points actually represent the distribution
				\item<5-> step size too large
			\end{itemize}
			\item<6-> \itshape{High Acceptance Rate:} $\gtrsim70\%$
			\begin{itemize}
				\item<7-> many points close to each other
				\item<8-> flat part of target distribution
				\item<9-> missing large portions of relevant parameter space
				\item<10-> step size too small
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Burn In}
		\begin{itemize}
			\item<2-> \itshape{Problem:}
			\begin{itemize}
				\item<3-> initialisation far from significant part of target
				\item<4-> flat distribution, majority of points not relevant
				\item<5-> no convergence to important porrtions of parameter space
			\end{itemize}
			\item<6-> \itshape{Fixes:}
			\begin{itemize}
				\item<7-> use ensemble of walkers
				\item<8-> identify burn in phase via trace plot and reject it
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Autocorrelation}
		\begin{itemize}
			\item<2-> \itshape{Definition:} $\displaystyle{\rho_t = \frac{\symup{Cov}(X_k \, , X_{k+t})}{\sqrt{\symup{Var}(X_k)\symup{Var}(X_{k+t})}} =
											 \frac{E[(X_k - E[X_k])(X_{k+t} - E[X_{k+t}])]}{\sqrt{E[(X_k - E[X_k])^2]E[(X_{k+t} - E[X_{k+t}])^2]}}}$
											 \vspace{1.5ex}
			\begin{itemize}
				\item<3-> Pearson correlation coefficient $\rho_t \in [-1, 1]$ for one parameter with a lagged version of itself
				\item<4-> Markov Chain elements are correlated with their neighbours since they are generated from their current states
				\item<5-> effective sample size is reduced by autocorrelation scale
				\item<6-> compute expectation values for $N$ samples by summing only to $M \ll N$ to avoid signal being dominated by statistical
						  noise and after discarding burn in
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}[noframenumbering]
		\frametitle{Autocorrelation}
		\begin{itemize}
			\item<2-> \itshape{Confidence:} $\displaystyle{\sigma_t^2 = \frac{1}{M} \left(1 + 2\sum_{s=1}^t\rho_s^2\right)}$\vspace{1.5ex}
			\begin{itemize}
				\item<3-> Bartlett estimate for uncertainty $\sigma_t$ of $\rho_t$ by a truly random process
				\item<4-> autocorrelation length $l$ is defined by lag $t$ after which $|\rho_t| < \sigma_t$ and therefore $\rho_t \neq 0$ no longer
						  statistically significant
			\end{itemize}
		\end{itemize}
	\end{frame}

	\subsection{Alternatives}

	\begin{frame}
		\frametitle{Shortcomings}
		\begin{itemize}
			\item<2-> \itshape{Expensive:} should only be used when absolutely necessary, do not do the following
			\begin{itemize}
				\item<3-> sample the entire parameter space
				\item<4-> improve poor optimisation performance
				\item<5-> discard distributions at intermediate steps
			\end{itemize}
			\item<6-> \itshape{Other Tools:}
			\begin{itemize}
				\item<7-> \itshape{Inversion Sampling:} transform uniform random numbers to desired distribution
				\item<8-> \itshape{Rejection Sampling:} propose samples, then accept or reject based on density
				\item<9-> \itshape{Importance Sampling:} weight samples from simple density to match target
			\end{itemize}
		\end{itemize}
	\end{frame}
