	\subsection{Fundamental Definitions}

	\begin{frame}
		\frametitle{Fluid Elements}
		\begin{itemize}
			\item<2-> \itshape{Fluids:} things that flow, conventionally liquids and gases, contrast solids with rigid lattice
			\item<3-> \itshape{Elements:} region with definable local variables, fulfilling certain conditions
			\begin{itemize}
				\item<4-> small enough to ignore systematic variations, size below any characteristic scale length
						  \visible<5->{$$\ell \ll \left|\frac{q}{\nabla q}\right|$$\\[-1ex]}
				\item<6-> large enough to ignore discreteness noise, statistically sufficient number of particles
						  \visible<7->{$$n\ell^3 \gg 1$$\\[-1ex]}
				\item<8-> in collisional fluids, large enough to communicate local conditions, requires interactions
						  \visible<9->{$$\ell \gg \lambda$$\\[-0.5ex]}
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Describing Fluid Equations}
		\begin{itemize}
			\item<2-> \itshape{Eulerian View:} cells of small volumes in fixed grid, flow is defined for all $t$ at
					  position $r$ with rate of change $\partial / \partial t$ for any measurable quantity $Q$
			\item<3-> \itshape{Lagrangian View:} follow particular fluid elements, comoving reference system with flow
					  evaluated in each one via $D / D t$ as the rate of change
			\item<4-> \itshape{Translating Relation:} add convective derivative term to account for movement of fluid element
					  \visible<5->{$$\frac{DQ}{Dt} = \frac{\partial Q}{\partial t} + u \cdot \nabla Q$$\\[-4.5ex]}
		\end{itemize}
	\end{frame}

	\subsection{Basic Equations}

	\begin{frame}
		\frametitle{Dimensionless Numbers}
		\begin{itemize}
			\item<2-> \itshape{Reynolds Number:} ratio of inertial to viscous forces
					  \visible<3->{$$R = \frac{\rho u L}{\eta} = \frac{u L}{\nu}$$\\[-1ex]}
			\begin{itemize}
				\item<3-> $R < 1000\kern+1pt${\bfseries{:}} Laminar Flow (reversible and ordered stream)
				\begin{itemize}
					\item<4-> $R < 1\kern+1pt${\bfseries{:}} Stokes Regime (without any separation)
					\item<5-> $10 \lesssim R \lesssim 40\kern+1pt${\bfseries{:}} Separation Zone (stable and stationary vortices)
				\end{itemize}
				\item<6-> $R > 1000\kern+1pt${\bfseries{:}} Turbulent Flow (stochastic and chaotic stream)
				\begin{itemize}
					\item<7-> $40 \lesssim R \lesssim 4000\kern+1pt${\bfseries{:}} Transitional Kármán Street (vortex shedding)
					\item<8-> $R > 4000\kern+1pt${\bfseries{:}} Kolmogorov Energy Cascade (fully developed phase)
				\end{itemize}
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}[noframenumbering]
		\frametitle{Dimensionless Numbers}
		\begin{itemize}
			\item<1-> \itshape{Mach Number:} ratio of bulk velocity magnitude to local sound speed
					  \visible<2->{$$M = \frac{u}{c}$$\\[-1ex]}
			\begin{itemize}
				\item<3-> $M \lesssim 0.3\kern+1pt${\bfseries{:}} Incompressible (negligible density variations, pressure signals faster than flow)
				\item<4-> $M \approx 1\kern+1pt${\bfseries{:}} Transonic (local excess of sound speed, weak shocks create drag)
				\item<5-> $M > 1\kern+1pt${\bfseries{:}} Supersonic (thin front of discontinuity, bow shock heating)
				\item<6-> $M > 3\kern+1pt${\bfseries{:}} Hypersonic (ram pressure stripping, dissociation and ionization)
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Bulk Transport}
		\begin{itemize}
			\item<2-> \itshape{Continuity Equation:} density $\rho$ must increase where a fluid is compressed
					  \visible<3->{$$\frac{\partial \rho}{\partial t} + \nabla \cdot (\rho u) = 0$$\\[-1ex]}
			\item<4-> \itshape{Linear Advection:} velocity field $v$ is free of divergence for incompressible fluids
					  \visible<5->{$$\frac{\partial u}{\partial t} + (v \cdot \nabla) u = 0$$\\[-1ex]}
			\item<6-> \itshape{Nonlinear Advection:} velocity field $u$ is carried by itself instead of fixed background flow
					  \visible<7->{$$\frac{\partial u}{\partial t} + (u \cdot \nabla) u = 0$$\\[-1ex]}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Equilibrium Search}
		\begin{itemize}
			\item<2-> \itshape{Laplace Equation:} maximally smooth solution dictated solely by boundary conditions
					  \visible<3->{$$\nabla^2 p = 0$$\\[-1ex]}
			\item<4-> \itshape{Poisson Equation:} extension for steady state to accomodate internal sources and sinks
					  \visible<5->{$$\nabla^2 p = f(r)$$\\[-1ex]}
			\begin{itemize}
				\item<6-> \itshape{Gravity:} $\nabla^2 \Phi = 4\pi G \rho$
				\item<7-> \itshape{Electrostatics:} $\nabla^2 V = -\rho / \varepsilon$
				\item<8-> \itshape{Pressure:} $\nabla^2 p = \rho\nabla\cdot\bigl((u\cdot\nabla) u\bigr)$
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Navier-Stokes Equations}
		\begin{itemize}
			\item<2-> \itshape{Momentum Equation:} ensure $\nabla \cdot u = 0$ by including $\nabla p$ to instantaneously cancel out any divergence
					  \visible<3->{$$\frac{\partial u}{\partial t} + (u \cdot \nabla) u = -\frac{1}{\rho} \nabla p + \nu \nabla^2 u$$\\[-1ex]}
			\item<4-> \itshape{Pressure Poisson:} require $p$ to have relaxed to its equilibrium for all $t$ before projecting
					  \visible<5->{$$\nabla^2 p = \rho\nabla\cdot\bigl((u\cdot\nabla) u\bigr)$$\\[-0.5ex]}
			\item<6-> fulfill at each step to accurately evolve initial configuration under given boundary conditions
		\end{itemize}
	\end{frame}

	\subsection{Finite Difference Method}

	\begin{frame}
		\frametitle{Conceptual Overview}
		\begin{itemize}
			\item<2-> \mdseries{Idea:} pointwise derivative estimation with series expansion via differences between adjacent nodes
			\item<3-> \mdseries{Advantages:}
				\begin{itemize}
					\item<4-> easy implementation
					\item<5-> computationally efficient
					\item<6-> highly accurate for simple problems
				\end{itemize}
			\item<7-> \mdseries{Drawbacks:} 
				\begin{itemize}
					\item<8-> struggles with complex geometries
					\item<9-> strong numerical diffusion at low orders
					\item<10-> inflexible rectilinear grid with fixed resolution
					\item<11-> mass and energy can leak due to numerical errors
				\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Integration Techniques}
		\begin{itemize}
			\item<2-> \itshape{Spatial:}
			\begin{itemize}
				\item<3-> \itshape{Downwind Scheme:} looks forward in direction of transport, unstable due to causality violation
				\item<4-> \itshape{Upwind Scheme:} looks backward against the direction of transport, very stable but diffusive
				\begin{itemize}
					\item<5-> used to model transport with defined direction
				\end{itemize}
				\item<6-> \itshape{Central Difference:} considers both directions, more precise but can lead to oscillations
				\begin{itemize}
					\item<7-> used to model symmetric dispersion processes
				\end{itemize}
				\item<8-> \itshape{Multi Point Stencil:} higher order central method for computing $\nabla^2$ in multiple dimensions
			\end{itemize}
			\item<9-> \itshape{Temporal:}
			\begin{itemize}
				\item<10-> \itshape{Backward Euler:} looks to future to calculate next step implicitly by inversion, extremely stable
				\item<11-> \itshape{Forward Euler:} looks to past to find next step explicitly, simple but unstable for large intervals
				\item<12-> \itshape{Leapfrog:} reaches back to previous step instead of current one, symplectic and invertible
			\end{itemize}
		\end{itemize}
	\end{frame}

	\begin{frame}[noframenumbering]
		\frametitle{Integration Techniques}
		\begin{itemize}
			\item<1-> \itshape{Advanced:}
			\begin{itemize}
				\item<2-> \itshape{Lax-Wendroff:} take intermediate step, equivalent to series expansion to second order
				\item<3-> \itshape{Runge-Kutta:} take trial steps to estimate curvature, then make informed jump
				\item<4-> \itshape{Predictor-Corrector:} take roughly predicted step, then recalculate with new information
				\item<5-> \itshape{Leapfrog:} preserve phasespace volume and total energy, enable backward reconstruction
			\end{itemize}
			\vspace{1ex}
			\item<6-> \itshape{Courant-Friedrichs-Lewy Condition:} $\displaystyle{\frac{v\Delta t}{\Delta x} \leq 1}$
			\item<8-> \itshape{Pure Diffusion Condition:} $\displaystyle{\frac{2\nu\Delta t}{\Delta x^2} \leq 1}$
		\end{itemize}
	\end{frame}

	\begin{frame}
		\frametitle{Discretized Formulation}
		\begin{itemize}
			\item<2-> \itshape{Convection:} $\displaystyle{\;\frac{\partial u}{\partial t} + u \frac{\partial u}{\partial x} = 0\;\longrightarrow\;
											 \frac{u_i^{n+1} - u_i^n}{\Delta t} + u_i^n \frac{u_i^n - u_{i-1}^n}{\Delta x} = 0}$
			\item<3-> \itshape{Diffusion:} $\displaystyle{\;\frac{\partial u}{\partial t} = \nu\frac{\partial^2 u}{\partial x^2}\;\longrightarrow\;
											\frac{u_i^{n+1} - u_i^n}{\Delta t} = \nu\frac{u_{i+1}^n - 2u_i^n + u_{i-1}^n}{\Delta x^2}}$
			\item<4-> \itshape{Bateman-Burgers Equation:} $\displaystyle{\; \frac{\partial u}{\partial t} + u \frac{\partial u}{\partial x} =
														   \nu\frac{\partial^2 u}{\partial x^2}\;\longrightarrow\;
														   \frac{u_i^{n+1} - u_i^n}{\Delta t} + u_i^n \frac{u_i^n - u_{i-1}^n}{\Delta x} =
														   \nu\frac{u_{i+1}^n - 2u_i^n + u_{i-1}^n}{\Delta x^2}}$
			\item<5-> \itshape{Laplace Equation:} $\displaystyle{\;\frac{\partial^2 p}{\partial x^2} + \frac{\partial^2 u}{\partial y^2} = 0\;
												   \longrightarrow\; \frac{p_{i+1, j}^n - 2u_{i, j}^n + u_{i-1, j}^n}{\Delta x^2} +
												   \frac{u_{i, j+1}^n - 2u_{i, j}^n + u_{i, j-1}^n}{\Delta y^2} = 0}$
		\end{itemize}
	\end{frame}

	\begin{frame}

				\begin{itemize}
